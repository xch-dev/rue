// The types of parameters are what the debug metadata says about their declared types, which the program doesn't check
// The program is the body of `main`
fn main(amount: Int, minimum: Int) {  // assert_message.rue:4:5-83
    if minimum > amount {             // assert_message.rue:4:5-83
        raise()                       // assert_message.rue:4:5-83
    } else {
        amount - minimum              // assert_message.rue:5:5-21
    }
}
===== debug =====
// The types of parameters are what the debug metadata says about their declared types, which the program doesn't check
program(solution) {                                // assert_message.rue:3:43-6:2
    main(amount: solution@2, minimum: solution@5)  // assert_message.rue:3:43-6:2
}

fn main(amount: Int, minimum: Int) {               // assert_message.rue:4:5-83
    // env@2 is always shortfall
    assert !(minimum > amount) else raise(         // assert_message.rue:4:5-83
        "assertion failed at assert_message.rue:4:5",
        ["amount is too small", shortfall(amount: amount, minimum: minimum)],
    )
    amount - minimum                               // assert_message.rue:5:5-21
}

fn shortfall(amount: Int, minimum: Int) {          // assert_message.rue:9:5-21
    minimum - amount                               // assert_message.rue:9:5-21
}
